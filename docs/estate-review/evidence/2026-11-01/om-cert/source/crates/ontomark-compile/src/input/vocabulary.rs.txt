523:         self.check_el_profile()?;
524:         for (key, def) in &self.relations {
525:             if let Some(inv) = &def.inverse {

561:     /// ADR-2126 Decision 2: refuse every entry whose OWL mapping or declared
562:     /// characteristics would put a non-EL construct into `ontology.ttl`.
563:     ///
564:     /// Checked sites are every IRI the emitter can write from the vocabulary:
565:     /// relation `owl` and `sub_property_of`, type and scalar `owl`, working
566:     /// extension `owl`, and super-property names, domains and ranges. A
567:     /// relation's `inverse:` is deliberately **not** checked: it names another
568:     /// relation key and is a build hint the emitter never turns into
569:     /// `owl:inverseOf`.
570:     fn check_el_profile(&self) -> std::result::Result<(), String> {
571:         let refuse = |site: String, iri: &str| -> std::result::Result<(), String> {
572:             let why = |term: &str| match term {
573:                 "hasValue" | "oneOf" => {
574:                     "which needs nominals, and Whelk does not implement nominals"
575:                 }
576:                 _ => "which is outside OWL 2 EL",
577:             };
578:             match self.non_el_term(iri) {
579:                 Some(term) => Err(format!(
580:                     "{NON_EL_VOCABULARY}: {site} maps to owl:{term}, {}; \
581:                      express the requirement as a `vault validate` rule (ADR-2126)",
582:                     why(term)
583:                 )),
584:                 None => Ok(()),
585:             }
586:         };
587:         for (key, def) in &self.types {
588:             refuse(format!("type {key:?} owl"), &def.owl)?;
589:         }
590:         for (key, def) in &self.relations {
591:             refuse(format!("relation {key:?} owl"), &def.owl)?;
592:             if let Some(sup) = &def.sub_property_of {
593:                 refuse(format!("relation {key:?} sub_property_of"), sup)?;
594:             }
595:             if let Some(d) = &def.domain {
596:                 refuse(format!("relation {key:?} domain"), d)?;
597:             }
598:             if let Some(r) = &def.range {
599:                 refuse(format!("relation {key:?} range"), r)?;
600:             }
601:             if def.characteristics.contains(&Characteristic::Symmetric) {
602:                 return Err(format!(
603:                     "{NON_EL_VOCABULARY}: relation {key:?} declares characteristic `symmetric`, \
604:                      which would emit owl:SymmetricProperty, outside OWL 2 EL (ADR-2126)"
605:                 ));
606:             }
607:         }
608:         for (key, def) in &self.super_properties {
609:             refuse(format!("super_property {key:?}"), key)?;
610:             if let Some(d) = &def.domain {
611:                 refuse(format!("super_property {key:?} domain"), d)?;
612:             }
613:             if let Some(r) = &def.range {
614:                 refuse(format!("super_property {key:?} range"), r)?;
615:             }
616:         }
617:         for (key, def) in &self.scalars {
618:             if let Some(owl) = &def.owl {
619:                 refuse(format!("scalar {key:?} owl"), owl)?;
620:             }
621:         }
622:         for (key, def) in &self.working_extensions {
623:             if let Some(owl) = &def.owl {
624:                 refuse(format!("working_extension {key:?} owl"), owl)?;
625:             }
626:         }
627:         Ok(())
628:     }
