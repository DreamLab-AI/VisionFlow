468: /// What `--repo` (or the current directory) turned out to be.
469: struct Repo {
470:     /// The repository root: the directory holding `ontology/` for an estate
471:     /// repository, or the vault itself for a plain one.
472:     root: PathBuf,
473:     /// The vocabulary found there, or the built-in one for a plain vault.
474:     vocab: Vocabulary,
475:     /// `true` when `ontology/vocabulary.yaml` was found: the estate shape.
476:     estate_shape: bool,
477: }
478:
479: /// Find the repository root: the nearest ancestor with
480: /// `ontology/vocabulary.yaml`. Without one, the start directory is a plain
481: /// Obsidian vault and is the root itself (ADR-OM-008). A vocabulary file that
482: /// exists but does not parse is still an error.
483: fn resolve_repo(explicit: Option<&Path>) -> anyhow::Result<Repo> {
484:     let start = match explicit {
485:         Some(p) => p.to_path_buf(),
486:         None => std::env::current_dir()?,
487:     };
488:     match Vocabulary::discover(&start) {
489:         Ok((path, vocab)) => {
490:             let root = path
491:                 .parent()
492:                 .and_then(Path::parent)
493:                 .map(Path::to_path_buf)
494:                 .unwrap_or(start);
495:             Ok(Repo {
496:                 root,
497:                 vocab,
498:                 estate_shape: true,
499:             })
500:         }
501:         Err(VaultError::NotAVault { .. }) => {
502:             anyhow::ensure!(
503:                 start.is_dir(),
504:                 "{} is not a directory (no ontology/vocabulary.yaml above it either)",
505:                 start.display()
506:             );
507:             Ok(Repo {
508:                 root: start,
509:                 vocab: Vocabulary::generic(),
510:                 estate_shape: false,
511:             })
512:         }
513:         Err(e) => Err(e).with_context(|| {
514:             format!(
515:                 "looking for ontology/vocabulary.yaml from {}",
516:                 start.display()
517:             )
518:         }),
519:     }
520: }

522: /// Resolve `--profile`: a built-in name or a YAML path.
523: ///
524: /// On a plain vault a built-in profile takes [`Layout::flat`]: the root is the
525: /// knowledge vault and there is no working vault. A profile YAML is taken as
526: /// written, `layout` included.
527: fn resolve_profile(spec: Option<&str>, repo: &Repo) -> anyhow::Result<Profile> {
528:     let mut profile = match spec {
529:         Some("estate") => ontomark_profile_estate::profile(),
530:         Some("generic") => Profile::generic(),
531:         Some(path) => {
532:             let text =
533:                 std::fs::read_to_string(path).with_context(|| format!("reading profile {path}"))?;
534:             return Profile::from_yaml(&text).with_context(|| format!("parsing profile {path}"));
535:         }
536:         None if repo.estate_shape => ontomark_profile_estate::profile(),
537:         None => Profile::generic(),
538:     };
539:     if !repo.estate_shape {
540:         profile.layout = Layout {
541:             excluded: profile.layout.excluded,
542:             ..Layout::flat()
543:         };
544:     }
545:     Ok(profile)
546: }
