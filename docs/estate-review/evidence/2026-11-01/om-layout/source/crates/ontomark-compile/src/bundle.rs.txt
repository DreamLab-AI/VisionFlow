1: //! Typed readers for the public C3 bundle and its scaffold/prose indexes.
2: //!
3: //! [`BundleManifest`] is the emitter's existing generation stamp, not a second
4: //! envelope. [`BundleManifest::from_json`] checks identity and safe artefact
5: //! claims before a consumer opens files. File bytes, digests, staleness and
6: //! activation still have to be checked by that consumer; deserialisation alone
7: //! never attests a served generation.
8: //!
9: //! Scaffold v1 omitted fields remain optional for existing readers. Omission
10: //! does not invent an ontology assertion or an attestation. Consumers bind the
11: //! three trust parameters to their own governance types; the compiler owns the
12: //! wire field names, while a hosting application owns trust interpretation.
13: //!
14: //! ```
15: //! use ontomark_compile::bundle::ScaffoldIndex;
16: //! let index: ScaffoldIndex = serde_json::from_str(
17: //!     r#"{"version":1,"generated":"2026-10-11","classes":{"a":{"t":"A"}}}"#
18: //! )?;
19: //! assert_eq!(index.classes["a"].t.as_deref(), Some("A"));
20: //! assert!(index.classes["a"].sup.is_empty());
21: //! # Ok::<(), serde_json::Error>(())
22: //! ```

37: /// A malformed or internally inconsistent manifest.
38: #[derive(Debug, thiserror::Error)]
39: pub enum ManifestError {
40:     /// JSON did not have the typed C3 shape, including every required field.
41:     #[error("invalid C3 manifest JSON: {0}")]
42:     Json(#[from] serde_json::Error),
43:     /// Identity, digest syntax, paths or artefact claims are inconsistent.
44:     #[error("invalid C3 manifest: {0}")]
45:     Invalid(String),
46: }
47:
48: impl Generation {
49:     /// Read and validate the current C3 marker without field coercion/defaults
50:     /// (bar `scope`, which is absent from a public marker and defaults to public).
51:     ///
52:     /// Unknown extension fields are permitted. This does not read or hash any
53:     /// artefact, consult a clock, fetch a resource or activate a generation.
54:     ///
55:     /// # Errors
56:     /// [`ManifestError`] for a malformed shape or failed [`Self::validate`].
57:     pub fn from_json(json: &str) -> Result<Self, ManifestError> {
58:         let manifest: Self = serde_json::from_str(json)?;
59:         manifest.validate()?;
60:         Ok(manifest)
61:     }
62:
63:     /// Check C3 identity, digest syntax, safe relative paths and unique claims.
64:     /// Source identity may be `unknown` outside Git, as the emitter specifies.
65:     ///
66:     /// The id is `<prefix>@<commit>[+dirty]` and the prefix is the producer's
67:     /// choice (`visionGraph` for the estate, a vault's directory name
68:     /// otherwise), so a reader needs no profile to check a marker: the prefix
69:     /// is read back from the id itself, everything before its last `@`, and
70:     /// must be non-empty; the rest must be exactly what
71:     /// [`generation_id`] spells for `commit` and `dirty`.
72:     ///
73:     /// # Errors
74:     /// [`ManifestError::Invalid`] when a required invariant is broken.
75:     pub fn validate(&self) -> Result<(), ManifestError> {
76:         let invalid = |reason: &str| ManifestError::Invalid(reason.to_owned());
77:         let consistent = self
78:             .id
79:             .rsplit_once('@')
80:             .filter(|(prefix, _)| !prefix.is_empty())
81:             .is_some_and(|(prefix, _)| self.id == generation_id(prefix, &self.commit, self.dirty));
82:         if self.commit.is_empty() || !consistent {
83:             return Err(invalid("id does not identify commit and dirty state"));
84:         }
85:         if !lower_hex(&self.content_digest, 64) {
86:             return Err(invalid("content_digest is not lowercase SHA-256"));
87:         }
88:         if !self
89:             .ontology_digest
90:             .strip_prefix("sha256-12-")
91:             .is_some_and(|s| lower_hex(s, 12))
92:         {
93:             return Err(invalid(
94:                 "ontology_digest is not a sha256-12 content address",
95:             ));
96:         }
97:         if self.generated_at.is_empty() {
98:             return Err(invalid("generated_at is empty"));
99:         }
100:         let mut names = HashSet::new();
101:         for artifact in &self.artifacts {
102:             let safe = !artifact.name.contains(['\\', '\0'])
103:                 && artifact
104:                     .name
105:                     .split('/')
106:                     .all(|p| !matches!(p, "" | "." | ".."));
107:             if !safe {
108:                 return Err(invalid("artefact name is not a safe relative POSIX path"));
109:             }
110:             if !names.insert(&artifact.name) {
111:                 return Err(invalid("artefact path is claimed more than once"));
112:             }
113:             if !lower_hex(&artifact.sha256, 64) {
114:                 return Err(invalid("artefact hash is not lowercase SHA-256"));
115:             }
116:         }
117:         Ok(())
118:     }
119: }
120:
121: fn lower_hex(value: &str, length: usize) -> bool {
122:     value.len() == length
123:         && value
124:             .bytes()
125:             .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
126: }
