650: /// How a build's generation id is spelled: `<prefix>@<commit>`, with `+dirty`
651: /// for an uncommitted build (see [`crate::build::generation::generation_id`]).
652: ///
653: /// With no `prefix` the repository root's directory name is used, so a vault
654: /// at `/vaults/notes` stamps `notes@<sha>`. The estate profile pins
655: /// `visionGraph`, which is what its consumers promote against.
656: ///
657: /// ```
658: /// use std::path::Path;
659: /// use ontomark_compile::profile::{GenerationNaming, Profile};
660: ///
661: /// let p = Profile::from_yaml("generation:\n  prefix: visionGraph\n").unwrap();
662: /// assert_eq!(p.generation.prefix_for(Path::new("/srv/anything")), "visionGraph");
663: /// assert_eq!(
664: ///     GenerationNaming::default().prefix_for(Path::new("/vaults/notes")),
665: ///     "notes"
666: /// );
667: /// ```
668: #[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
669: #[serde(default, deny_unknown_fields)]
670: pub struct GenerationNaming {
671:     /// The text before `@` in every generation id. `None` uses the repository
672:     /// root's directory name.
673:     pub prefix: Option<String>,
674: }
675:
676: impl GenerationNaming {
677:     /// The prefix generation ids under this profile carry for the repository
678:     /// at `repo_root`: [`GenerationNaming::prefix`] when set, else the root's
679:     /// directory name (after resolving `.` and symlinks), else `ontomark` for
680:     /// a root with no name at all (`/`).
681:     #[must_use]
682:     pub fn prefix_for(&self, repo_root: &Path) -> String {
683:         if let Some(prefix) = self.prefix.as_deref().filter(|p| !p.is_empty()) {
684:             return prefix.to_owned();
685:         }
686:         let resolved = std::fs::canonicalize(repo_root).unwrap_or_else(|_| repo_root.to_path_buf());
687:         resolved
688:             .file_name()
689:             .and_then(|n| n.to_str())
690:             .filter(|n| !n.is_empty())
691:             .map_or_else(|| "ontomark".to_owned(), str::to_owned)
692:     }
693: }
