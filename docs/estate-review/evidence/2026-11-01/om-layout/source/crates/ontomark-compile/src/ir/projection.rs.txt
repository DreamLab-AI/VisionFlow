86: /// [`inspect_inputs`] over a walk the caller already made, so a build walks
87: /// the vault root once for the census and the load (OM-00 step 1).
88: ///
89: /// The census and the vault load share one listing and one skip rule: a
90: /// hidden-directory or excluded-folder skip counts as `excluded_directory`, a
91: /// non-regular `.md` entry as `excluded_not_regular_file`, and both are
92: /// visible in the counts.
93: ///
94: /// # Errors
95: /// As [`inspect_inputs`].
96: pub fn inspect_walk(
97:     root: &Path,
98:     walk: &crate::input::page::PageWalk,
99: ) -> Result<Census, PublicationBlocked> {
100:     use crate::input::page::SkipReason;
101:     let mut counts: BTreeMap<String, usize> = BTreeMap::new();
102:     let mut bump = |k: &str| *counts.entry(k.to_owned()).or_insert(0) += 1;
103:
104:     for skip in &walk.skipped {
105:         bump("input_files");
106:         bump(match skip.reason {
107:             SkipReason::HiddenDirectory | SkipReason::ExcludedDirectory => "excluded_directory",
108:             SkipReason::NotARegularFile => "excluded_not_regular_file",
109:         });
110:     }
111:     for path in &walk.files {
112:         bump("input_files");
113:         let rel = path.strip_prefix(root).unwrap_or(path);
114:         if is_unpublished(rel) {
115:             bump("excluded_directory");
116:             continue;
117:         }
118:         let Ok(text) = std::fs::read_to_string(path) else {
119:             bump("invalid_input");
120:             continue;
121:         };
122:         let split = frontmatter::split(&text);
123:         if split.unterminated {
124:             // An opening fence with no close is a malformed page, not a
125:             // private one (S3.3).
126:             bump("invalid_input");
127:             continue;
128:         }
129:         let Some(yaml) = split.yaml else {
130:             bump("excluded_no_frontmatter");
131:             continue;
132:         };
133:         let Ok(fm) = frontmatter::Frontmatter::parse(yaml) else {
134:             bump("invalid_input");
135:             continue;
136:         };
137:         // Never coerce a string or a number into publication permission: only a
138:         // real YAML boolean counts, and anything else refuses the build.
139:         match fm.get("public") {
140:             Some(serde_yaml::Value::Bool(true)) => bump("included_public"),
141:             None | Some(serde_yaml::Value::Bool(false)) => bump("included_private"),
142:             Some(_) => bump("invalid_publication_flag"),
143:         }
144:     }
145:
146:     let invalid: usize = counts
147:         .iter()
220:     }
221:     Ok(out)
222: }
