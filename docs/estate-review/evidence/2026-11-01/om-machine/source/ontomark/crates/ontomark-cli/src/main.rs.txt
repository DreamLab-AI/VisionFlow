566:         .collect()
790:                 }

797:             // the indexes and the graph tiers are `knowledge/`'s by definition.
798:             // `--vault all` widens the `publish/` staging to `working/`'s
799:             // `public: true` pages, which is the only thing "both vaults" can
800:             // mean for a build.

802:             // A machine bundle always reads both vaults from the knowledge
803:             // root, so `--vault working|all` is left for the build to refuse
804:             // by name rather than quietly moving the root.
805:             let machine = args.scope == ScopeArg::Machine;
806:             let kind = if kinds == [VaultKind::Working] && !machine {
807:                 VaultKind::Working
808:             } else {
809:                 VaultKind::Knowledge
810:             };

812:                 with_working_publish: kinds.contains(&VaultKind::Working)
813:                     && kind == VaultKind::Knowledge,
814:                 publish_out: args.publish_out.clone(),
815:                 vault_root: vault_root(&root, &profile, kind)?,
816:                 out: args.out.clone(),

818:                 with_markdown_mirror: args.with_markdown_mirror,
819:                 generated_at: args.generated_at.clone(),
820:                 source: args

822:                     .clone()
