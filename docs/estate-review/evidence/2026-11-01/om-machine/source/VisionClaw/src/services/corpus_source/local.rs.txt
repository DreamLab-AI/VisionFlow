44: /// Walks a vault on disk: every `.md` file under each configured base path,
45: /// keyed on `mtime:size` so an unchanged file is skipped by the incremental
46: /// filter without reading it.
47: #[derive(Debug, Clone)]
48: pub struct LocalDirectorySource {
49:     /// Absolute (or process-relative) path to the vault root.
50:     pub root: PathBuf,
51:     /// Ingest base paths relative to [`Self::root`], in order.
52:     pub base_paths: Vec<PathBuf>,
53:     /// The same paths rendered with `/` separators, built once at
54:     /// construction so [`CorpusSource::base_paths`] can hand out a slice.
55:     slash_paths: Vec<String>,
56: }
57:
58: impl LocalDirectorySource {
59:     /// Construct from an explicit root and base paths.
60:     pub fn new(root: impl Into<PathBuf>, base_paths: Vec<PathBuf>) -> Self {
61:         let slash_paths = base_paths.iter().map(|p| path_to_slash(p)).collect();
62:         Self {
63:             root: root.into(),
64:             base_paths,
65:             slash_paths,
66:         }
67:     }
68:
69:     /// Construct from the environment: `VAULT_ROOT` plus `VAULT_BASE_PATHS`
70:     /// (default `knowledge,working`).
71:     ///
72:     /// Fails when `VAULT_ROOT` is unset or empty — the local source cannot
73:     /// guess where the vault is mounted.
74:     pub fn from_env() -> Result<Self, String> {
75:         let root = std::env::var(VAULT_ROOT_ENV)
76:             .ok()
77:             .map(|v| v.trim().to_string())
78:             .filter(|v| !v.is_empty())
79:             .ok_or_else(|| {
80:                 format!(
81:                     "{} is not set — the local corpus source needs the vault mount point",
82:                     VAULT_ROOT_ENV
83:                 )
84:             })?;
85:
86:         Ok(Self::new(root, Self::base_paths_from_env()?))
87:     }
