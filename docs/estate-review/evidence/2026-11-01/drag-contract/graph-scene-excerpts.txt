3388: ## Starts `hand`'s grab of node `node_id` in volume `vid`: records the hand's
3389: ## state, selects the node, and sends nodeDragStart on that volume's connection.
3390: ## `hand_pos` is the controller's world position (grab depth is measured from
3391: ## it); `hand_basis` (the controller's global basis) lets the drive hold the node
3392: ## at the same hand-local offset it was caught at, so an off-axis catch does not snap
3393: ## it onto the ray. False and no state change when the grab is refused: the hand already
3394: ## holds a node, the other hand holds this one, the volume's connection has no
3395: ## DRAG (codescape), the volume is a portal of the entered space, or a room
3396: ## transition is in flight.
3397: func _begin_grab(hand: int, vid: StringName, node_id: int, hand_pos: Vector3, hand_basis: Variant = null) -> bool:
3398: 	if hand < 0 or hand > 1 or node_id < 0 or _grab_id[hand] != -1 or _is_held(node_id):
3399: 		return false
3400: 	if not volume_serves_drag(vid) or not grab_volume_allowed(vid, _entered_volume()) or _grab_locked():
3401: 		return false
3402: 	var conn: Node = _conn_for_volume(vid)
3403: 	if conn == null:
3404: 		return false
3405: 	_grab_id[hand] = node_id
3406: 	_grab_vol[hand] = vid
3407: 	_sync_grab_exclusion(vid)
3408: 	print("GraphScene DEBUG: node_grabbed hand=%d volume=%s id=%d" % [hand, vid, node_id])
3409: 	# The held node is the selection (UX feature 2): the Rust tube may catch a
3410: 	# node the volume picker's cone missed, or a different one.
3411: 	select_volume_node(vid, StringName(VOLUME_BINDINGS[vid]["conn"]), node_id)
3412: 	# Node render position lives in the connection's Rust store.
3413: 	var pos: Vector3 = Vector3.ZERO
3414: 	if conn.client != null and conn.client.has_method("node_position"):
3415: 		pos = conn.client.node_position(node_id)
3416: 	_grab_target[hand] = pos
3417: 	# Remember how far along the ray the node was so the drag preserves depth.
3418: 	var node_world: Vector3 = _content_xf(vid) * pos
3419: 	_grab_dist[hand] = clampf(hand_pos.distance_to(node_world), 0.2, 6.0)
3420: 	_grab_lateral[hand] = Vector3.ZERO
3421: 	if hand_basis is Basis:
3422: 		# The catch offset in the hand's own frame: depth along -Z, lateral in x / y.
3423: 		var local: Vector3 = (hand_basis as Basis).orthonormalized().inverse() * (node_world - hand_pos)
3424: 		_grab_lateral[hand] = Vector3(local.x, local.y, 0.0)
